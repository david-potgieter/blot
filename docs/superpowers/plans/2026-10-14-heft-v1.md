# heft v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Go TUI that scans macOS folders and shows files/folders over a size threshold as a navigable tree.

**Architecture:** `scan` walks roots in parallel (fastwalk) into a dir tree, pruning small files into parent totals. `tree` is pure filtering/sorting/flattening of that tree into rows. `ui` is a Bubble Tea v2 state machine: huh form → scanning screen → tree screen. `cmd/heft` parses flags.

**Tech Stack:** Go 1.26, charm.land/bubbletea/v2 v2.1.0, charm.land/bubbles/v2 v2.2.1, charm.land/lipgloss/v2 v2.0.6, charm.land/huh/v2 v2.0.3, github.com/charlievieth/fastwalk v1.0.14, github.com/dustin/go-humanize v1.1.0.

Spec: `docs/superpowers/specs/2026-10-14-heft-design.md`.
Deviation from spec: UI tests drive models directly (`Update` + `View` assertions) instead of teatest golden files — less brittle, same coverage.

```go
// Package size parses and formats human-readable byte sizes.
package size

import (
	"fmt"
	"math"
	"strings"

	"github.com/dustin/go-humanize"
)

// Parse converts strings like "500MB", "1.5GiB" or "1024" into bytes.
func Parse(s string) (int64, error) {
	n, err := humanize.ParseBytes(strings.TrimSpace(s))
	if err != nil {
		return 0, fmt.Errorf("invalid size %q (try 500MB or 1GiB)", s)
	}
	if n > math.MaxInt64 {
		return 0, fmt.Errorf("size %q too large", s)
	}
	return int64(n), nil
}

// Format renders bytes with SI units, e.g. 1500000 → "1.5 MB".
func Format(n int64) string {
	if n < 0 {
		n = 0
	}
	return humanize.Bytes(uint64(n))
}
```

```go
// Package scan walks directories and builds a size-annotated tree.
package scan

import (
	"context"
	"io/fs"
	"os"
	"path/filepath"
	"sync"
	"sync/atomic"
	"syscall"
	"time"

	"github.com/charlievieth/fastwalk"
)

const maxErrors = 100

// Node is a file or directory. Dir sizes are recursive disk usage.
type Node struct {
	Name     string
	Path     string
	IsDir    bool
	Size     int64 // bytes on disk (st_blocks*512)
	ModTime  time.Time
	Parent   *Node
	Children []*Node // dirs: all subdirs + files with Size >= Options.FileFloor
}

// Options control a scan.
type Options struct {
	FileFloor int64    // files smaller than this count toward parent size but get no Node
	Exclude   []string // absolute paths to skip
}

// Progress is a live snapshot of scan counters.
type Progress struct {
	Files, Dirs, Bytes, Errors int64
	Current                    string
}

// Result is a finished scan.
type Result struct {
	Roots                         []*Node
	Files, Dirs, Bytes, ErrorCount int64
	Errors                        []error // first maxErrors errors
}

type inode struct{ dev, ino uint64 }

type scanner struct {
	opts     Options
	excluded map[string]bool

	files, dirs, bytes, errCount atomic.Int64
	current                      atomic.Pointer[string]

	mu     sync.Mutex
	errs   []error
	nodes  map[string]*Node // dir path → node, per root
	linked map[inode]bool   // hard links seen, per root
}

// DefaultExcludes returns paths skipped by default on macOS.
func DefaultExcludes(home string) []string {
	return []string{
		"/System",
		"/Volumes",
		"/dev",
		"/private/var/vm",
		filepath.Join(home, "Library", "Mobile Documents"),
		filepath.Join(home, ".Trash"),
	}
}

// Scan walks each root and returns its tree. Progress snapshots are sent
// non-blocking to progress (may be nil) roughly 10×/s. Permission and read
// errors are collected, not returned; only a bad root or cancellation fails.
func Scan(ctx context.Context, roots []string, opts Options, progress chan<- Progress) (*Result, error) {
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	s := &scanner{opts: opts, excluded: map[string]bool{}}
	for _, e := range opts.Exclude {
		s.excluded[filepath.Clean(e)] = true
	}

	stop := make(chan struct{})
	var wg sync.WaitGroup
	if progress != nil {
		wg.Add(1)
		go func() {
			defer wg.Done()
			t := time.NewTicker(100 * time.Millisecond)
			defer t.Stop()
			for {
				select {
				case <-stop:
					return
				case <-t.C:
					s.emit(progress)
				}
			}
		}()
	}

	res := &Result{}
	var err error
	for _, root := range roots {
		var n *Node
		if n, err = s.scanRoot(ctx, root); err != nil {
			break
		}
		res.Roots = append(res.Roots, n)
	}
	close(stop)
	wg.Wait()
	if err != nil {
		return nil, err
	}
	s.emit(progress)

	res.Files, res.Dirs, res.Bytes, res.ErrorCount = s.files.Load(), s.dirs.Load(), s.bytes.Load(), s.errCount.Load()
	res.Errors = s.errs
	return res, nil
}

func (s *scanner) scanRoot(ctx context.Context, root string) (*Node, error) {
	abs, err := filepath.Abs(root)
	if err != nil {
		return nil, err
	}
	real, err := filepath.EvalSymlinks(abs)
	if err != nil {
		return nil, err
	}
	info, err := os.Lstat(real)
	if err != nil {
		return nil, err
	}
	st := info.Sys().(*syscall.Stat_t)
	if !info.IsDir() {
		n := &Node{Name: real, Path: real, Size: st.Blocks * 512, ModTime: info.ModTime()}
		s.files.Add(1)
		s.bytes.Add(n.Size)
		return n, nil
	}

	s.nodes = map[string]*Node{}
	s.linked = map[inode]bool{}
	if err := fastwalk.Walk(&fastwalk.Config{}, real, s.walkFn(ctx, real, uint64(st.Dev))); err != nil {
		return nil, err
	}
	n := s.nodes[real]
	sumDirs(n)
	return n, nil
}

func (s *scanner) walkFn(ctx context.Context, root string, dev uint64) fs.WalkDirFunc {
	return func(path string, d fs.DirEntry, err error) error {
		if ctxErr := ctx.Err(); ctxErr != nil {
			return ctxErr
		}
		if err != nil {
			s.addErr(err)
			return nil
		}
		path = filepath.Clean(path)
		if path != root && s.excluded[path] {
			if d.IsDir() {
				return fs.SkipDir
			}
			return nil
		}
		info, err := d.Info()
		if err != nil {
			s.addErr(err)
			return nil
		}
		st, ok := info.Sys().(*syscall.Stat_t)
		if !ok {
			return nil
		}
		usage := st.Blocks * 512
		if d.IsDir() {
			if uint64(st.Dev) != dev {
				return fs.SkipDir // different filesystem (mount point)
			}
			s.addDir(path, root, info.ModTime(), usage)
			return nil
		}
		if st.Nlink > 1 && !s.firstLink(inode{uint64(st.Dev), st.Ino}) {
			return nil
		}
		s.addFile(path, info.ModTime(), usage)
		return nil
	}
}

func (s *scanner) addDir(path, root string, mod time.Time, usage int64) {
	n := &Node{Name: filepath.Base(path), Path: path, IsDir: true, Size: usage, ModTime: mod}
	s.dirs.Add(1)
	s.current.Store(&path)

	s.mu.Lock()
	defer s.mu.Unlock()
	s.nodes[path] = n
	if path == root {
		n.Name = path
		return
	}
	if p := s.nodes[filepath.Dir(path)]; p != nil {
		n.Parent = p
		p.Children = append(p.Children, n)
	}
}

func (s *scanner) addFile(path string, mod time.Time, usage int64) {
	s.files.Add(1)
	s.bytes.Add(usage)

	s.mu.Lock()
	defer s.mu.Unlock()
	p := s.nodes[filepath.Dir(path)]
	if p == nil {
		return
	}
	p.Size += usage
	if usage >= s.opts.FileFloor {
		p.Children = append(p.Children, &Node{Name: filepath.Base(path), Path: path, Size: usage, ModTime: mod, Parent: p})
	}
}

func (s *scanner) firstLink(k inode) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.linked[k] {
		return false
	}
	s.linked[k] = true
	return true
}

func (s *scanner) addErr(err error) {
	s.errCount.Add(1)
	s.mu.Lock()
	defer s.mu.Unlock()
	if len(s.errs) < maxErrors {
		s.errs = append(s.errs, err)
	}
}

func (s *scanner) emit(ch chan<- Progress) {
	if ch == nil {
		return
	}
	p := Progress{Files: s.files.Load(), Dirs: s.dirs.Load(), Bytes: s.bytes.Load(), Errors: s.errCount.Load()}
	if c := s.current.Load(); c != nil {
		p.Current = *c
	}
	select {
	case ch <- p:
	default:
	}
}

// sumDirs adds subdirectory totals into each dir, bottom-up.
func sumDirs(n *Node) int64 {
	for _, c := range n.Children {
		if c.IsDir {
			n.Size += sumDirs(c)
		}
	}
	return n.Size
}
```

```go
// Package tree filters, sorts and flattens a scan tree into display rows.
package tree

import (
	"cmp"
	"fmt"
	"slices"
	"strings"

	"github.com/david-potgieter/heft/internal/scan"
)

// Type selects which kinds of nodes match.
type Type int

const (
	All Type = iota
	Dirs
	Files
)

var typeNames = [...]string{"all", "dirs", "files"}

func (t Type) String() string { return typeNames[t] }

// ParseType parses "all", "dirs" or "files".
func ParseType(s string) (Type, error) {
	for i, n := range typeNames {
		if strings.EqualFold(s, n) {
			return Type(i), nil
		}
	}
	return All, fmt.Errorf("invalid type %q (want all, dirs or files)", s)
}

// Filter decides which nodes match.
type Filter struct {
	Min  int64
	Type Type
}

// Matches reports whether n satisfies the size and type constraints.
func (f Filter) Matches(n *scan.Node) bool {
	if n.Size < f.Min {
		return false
	}
	switch f.Type {
	case Dirs:
		return n.IsDir
	case Files:
		return !n.IsDir
	}
	return true
}

// SortBy orders siblings.
type SortBy int

const (
	BySize SortBy = iota
	ByName
)

func (s SortBy) String() string {
	if s == ByName {
		return "name"
	}
	return "size"
}

// Row is one visible line. Context rows are shown only to reach matches.
type Row struct {
	Node        *scan.Node
	Depth       int
	Expanded    bool
	HasChildren bool
	Context     bool
}

// Tree holds visibility, sort and expansion state over scan roots.
type Tree struct {
	roots    []*scan.Node
	filter   Filter
	sort     SortBy
	kids     map[*scan.Node][]*scan.Node // visible children, sorted
	expanded map[*scan.Node]bool
	matches  int
}

// New computes visibility once. Roots are always shown and start expanded.
func New(roots []*scan.Node, f Filter) *Tree {
	t := &Tree{roots: roots, filter: f, kids: map[*scan.Node][]*scan.Node{}, expanded: map[*scan.Node]bool{}}
	for _, r := range roots {
		t.visit(r)
		t.expanded[r] = true
	}
	t.sortAll()
	return t
}

// visit caches visible children of n and reports whether n is visible.
// Sizes are monotonic up the tree, so nothing under a node below Min can match.
func (t *Tree) visit(n *scan.Node) bool {
	if n.Size < t.filter.Min {
		return false
	}
	var kids []*scan.Node
	for _, c := range n.Children {
		if t.visit(c) {
			kids = append(kids, c)
		}
	}
	if len(kids) > 0 {
		t.kids[n] = kids
	}
	if t.filter.Matches(n) {
		t.matches++
		return true
	}
	return n.IsDir && len(kids) > 0
}

func (t *Tree) Filter() Filter   { return t.filter }
func (t *Tree) Sort() SortBy     { return t.sort }
func (t *Tree) MatchCount() int  { return t.matches }
func (t *Tree) Roots() []*scan.Node { return t.roots }

// SetSort reorders all visible children.
func (t *Tree) SetSort(s SortBy) {
	t.sort = s
	t.sortAll()
}

func (t *Tree) sortAll() {
	for _, k := range t.kids {
		slices.SortStableFunc(k, t.compare)
	}
}

func (t *Tree) compare(a, b *scan.Node) int {
	if t.sort == ByName {
		return cmp.Compare(strings.ToLower(a.Name), strings.ToLower(b.Name))
	}
	if c := cmp.Compare(b.Size, a.Size); c != 0 {
		return c
	}
	return cmp.Compare(strings.ToLower(a.Name), strings.ToLower(b.Name))
}

// Expand opens n if it has visible children.
func (t *Tree) Expand(n *scan.Node) {
	if len(t.kids[n]) > 0 {
		t.expanded[n] = true
	}
}

// Collapse closes n.
func (t *Tree) Collapse(n *scan.Node) { delete(t.expanded, n) }

// Rows flattens visible, expanded nodes depth-first.
func (t *Tree) Rows() []Row {
	var rows []Row
	var walk func(n *scan.Node, depth int)
	walk = func(n *scan.Node, depth int) {
		kids := t.kids[n]
		exp := t.expanded[n] && len(kids) > 0
		rows = append(rows, Row{Node: n, Depth: depth, Expanded: exp, HasChildren: len(kids) > 0, Context: !t.filter.Matches(n)})
		if exp {
			for _, c := range kids {
				walk(c, depth+1)
			}
		}
	}
	for _, r := range t.roots {
		walk(r, 0)
	}
	return rows
}
```

```go
package ui

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/david-potgieter/heft/internal/size"
	"github.com/david-potgieter/heft/internal/tree"
)

// Params is a validated search.
type Params struct {
	Roots []string
	Min   int64
	Type  tree.Type
}

func (p Params) summary() string {
	home, _ := os.UserHomeDir()
	roots := make([]string, len(p.Roots))
	for i, r := range p.Roots {
		roots[i] = tildify(r, home)
	}
	return fmt.Sprintf("%s · ≥ %s · %s", strings.Join(roots, ", "), size.Format(p.Min), p.Type)
}

// formValues backs the huh form fields (huh needs stable pointers).
type formValues struct {
	roots string
	min   string
	typ   tree.Type
}

func (v *formValues) params() (Params, error) {
	roots, err := parseRoots(v.roots)
	if err != nil {
		return Params{}, err
	}
	min, err := size.Parse(v.min)
	if err != nil {
		return Params{}, err
	}
	return Params{Roots: roots, Min: min, Type: v.typ}, nil
}

// parseRoots splits comma-separated paths, expands ~ and checks they exist.
func parseRoots(s string) ([]string, error) {
	home, _ := os.UserHomeDir()
	var out []string
	for _, p := range strings.Split(s, ",") {
		p = strings.TrimSpace(p)
		if p == "" {
			continue
		}
		if p == "~" {
			p = home
		} else if strings.HasPrefix(p, "~/") {
			p = filepath.Join(home, p[2:])
		}
		if _, err := os.Stat(p); err != nil {
			return nil, fmt.Errorf("%s: not found", p)
		}
		out = append(out, filepath.Clean(p))
	}
	if len(out) == 0 {
		return nil, errors.New("enter at least one folder")
	}
	return out, nil
}

func tildify(p, home string) string {
	if home != "" && (p == home || strings.HasPrefix(p, home+"/")) {
		return "~" + p[len(home):]
	}
	return p
}
```

```go
	offset        int

	m := treeModel{
		tree:   tree.New(res.Roots, tree.Filter{Min: p.Min, Type: p.Type}),
		res:    res,
		params: p,

		styles: st,
	}
	m.help.SetWidth(w)
	m.refresh()
	return m
}

func (m *treeModel) refresh() {
	m.rows = m.tree.Rows()
	m.cursor = max(0, min(m.cursor, len(m.rows)-1))
}

func (m treeModel) current() *scan.Node { return m.rows[m.cursor].Node }

func (m treeModel) Update(msg tea.Msg) (treeModel, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
		m.help.SetWidth(msg.Width)
	case tea.KeyPressMsg:
		switch {
		case key.Matches(msg, m.keys.Quit):
			return m, tea.Quit
		case key.Matches(msg, m.keys.Up):
			m.cursor--
		case key.Matches(msg, m.keys.Down):
			m.cursor++
		case key.Matches(msg, m.keys.Top):
			m.cursor = 0
		case key.Matches(msg, m.keys.Bottom):
			m.cursor = len(m.rows) - 1
		case key.Matches(msg, m.keys.Expand):
			m.tree.Expand(m.current())
		case key.Matches(msg, m.keys.Collapse):
			m.collapse()
		case key.Matches(msg, m.keys.Sort):
			if m.tree.Sort() == tree.BySize {
				m.tree.SetSort(tree.ByName)
			} else {
				m.tree.SetSort(tree.BySize)
			}
		case key.Matches(msg, m.keys.Reveal):
			return m, reveal(m.current().Path)
		case key.Matches(msg, m.keys.Rescan):
			return m, func() tea.Msg { return rescanMsg{} }
		case key.Matches(msg, m.keys.New):
			return m, func() tea.Msg { return newSearchMsg{} }
		case key.Matches(msg, m.keys.Help):
			m.help.ShowAll = !m.help.ShowAll
		}
		m.refresh()
	}
	return m, nil
}

// collapse closes the current row, or jumps to its parent row.

		m.tree.Collapse(row.Node)
		return
	}
	for i := m.cursor - 1; i >= 0; i-- {
		if m.rows[i].Node == row.Node.Parent {
			m.cursor = i
			return
		}
	}
}

		size.Format(m.res.Bytes), humanize.Comma(m.res.Files), m.tree.MatchCount())

		footer += " · " + s.err.Render(fmt.Sprintf("%d errors — grant your terminal Full Disk Access for complete results", m.res.ErrorCount))
	}

	}
	end := min(len(m.rows), offset+listHeight)

	var b strings.Builder
	b.WriteString(header + "\n\n")
	for i := offset; i < end; i++ {
		b.WriteString(m.renderRow(m.rows[i], i == m.cursor) + "\n")
	}
	for i := end - offset; i < listHeight; i++ {
		b.WriteString("\n")
	}
	b.WriteString(footer + "\n" + helpView)
	return b.String()
}

func (m treeModel) renderRow(r tree.Row, selected bool) string {
	s := m.styles
	n := r.Node

	marker := "  "
	if r.HasChildren {
		marker = "▸ "
		if r.Expanded {
			marker = "▾ "
		}
	}
	name := n.Name
	if n.IsDir && r.Depth > 0 {
		name += "/"
	}
	nameStyle := lipgloss.NewStyle()
	if r.Context {
		nameStyle = s.context
	}

	line := fmt.Sprintf("%s %s %s%s%s  %s",
		s.size.Render(fmt.Sprintf("%9s", size.Format(n.Size))),
		m.bar(n),
		strings.Repeat("  ", r.Depth),
		s.marker.Render(marker),
		nameStyle.Render(name),
		s.subtle.Render(humanize.Time(n.ModTime)),
	)
	if selected {
		line = s.selected.Render(line)
	}
	if m.width > 0 {
		line = lipgloss.NewStyle().MaxWidth(m.width).Render(line)
	}
	return line
}

// bar shows n's share of its parent (roots are full).
func (m treeModel) bar(n *scan.Node) string {
	filled := barWidth
	if p := n.Parent; p != nil && p.Size > 0 {
		filled = int(float64(n.Size)/float64(p.Size)*barWidth + 0.5)
	}
	filled = max(0, min(barWidth, filled))
	return m.styles.bar.Render(strings.Repeat("█", filled)) + m.styles.barEmpty.Render(strings.Repeat("░", barWidth-filled))
}
```

Note: `offset` isn't persisted from `View` (value receiver); scrolling recomputes from cursor each render, which keeps the cursor in view. Acceptable for v1.

```go
package ui

import (
	"context"
	"fmt"
	"time"

	"charm.land/bubbles/v2/spinner"
	tea "charm.land/bubbletea/v2"
	"github.com/dustin/go-humanize"

	"github.com/david-potgieter/heft/internal/scan"
	"github.com/david-potgieter/heft/internal/size"
)

// Messages carry the scan id so results from cancelled scans are ignored.
type progressMsg struct {
	id int
	p  scan.Progress
}

type scanDoneMsg struct {
	id  int
	res *scan.Result
	err error
}

type scanModel struct {
	id       int
	params   Params
	spinner  spinner.Model
	progress scan.Progress
	cancel   context.CancelFunc
	ch       chan scan.Progress
	done     chan scanDoneMsg
	started  time.Time
	styles   styles
}

// newScanModel starts the scan in the background.
func newScanModel(id int, p Params, excludes []string, st styles) scanModel {
	ctx, cancel := context.WithCancel(context.Background())
	m := scanModel{
		id:      id,
		params:  p,
		spinner: spinner.New(spinner.WithSpinner(spinner.Dot), spinner.WithStyle(st.title)),
		cancel:  cancel,
		ch:      make(chan scan.Progress, 1),
		done:    make(chan scanDoneMsg, 1),
		started: time.Now(),
		styles:  st,
	}
	go func() {
		res, err := scan.Scan(ctx, p.Roots, scan.Options{FileFloor: p.Min, Exclude: excludes}, m.ch)
		m.done <- scanDoneMsg{id: id, res: res, err: err}
	}()
	return m
}

func (m scanModel) Init() tea.Cmd { return tea.Batch(m.spinner.Tick, m.wait()) }

func (m scanModel) wait() tea.Cmd {
	return func() tea.Msg {
		select {
		case p := <-m.ch:
			return progressMsg{id: m.id, p: p}
		case d := <-m.done:
			return d
		}
	}
}

func (m scanModel) Update(msg tea.Msg) (scanModel, tea.Cmd) {
	switch msg := msg.(type) {
	case progressMsg:
		if msg.id != m.id {
			return m, nil
		}
		m.progress = msg.p
		return m, m.wait()
	case spinner.TickMsg:
		var cmd tea.Cmd
		m.spinner, cmd = m.spinner.Update(msg)
		return m, cmd
	}
	return m, nil
}

func (m scanModel) View(width int) string {
	s := m.styles
	p := m.progress
	cur := p.Current
	if width > 4 && len(cur) > width-4 {
		cur = "…" + cur[len(cur)-(width-5):]
	}
	return fmt.Sprintf("%s %s %s\n\n  %s files · %s folders · %s · %s errors · %s\n\n  %s\n\n  %s",
		m.spinner.View(), s.title.Render("Scanning"), s.subtle.Render(m.params.summary()),
		humanize.Comma(p.Files), humanize.Comma(p.Dirs), size.Format(p.Bytes), humanize.Comma(p.Errors),
		time.Since(m.started).Round(time.Second),
		s.subtle.Render(cur),
		s.subtle.Render("esc cancel · ctrl+c quit"),
	)
}
```

```go
// Package ui is heft's Bubble Tea interface.
package ui

import (
	"context"
	"errors"

	tea "charm.land/bubbletea/v2"
	"charm.land/huh/v2"

	"github.com/david-potgieter/heft/internal/tree"
)

type state int

const (
	stateForm state = iota
	stateScanning
	stateTree
)

// Config holds initial form values (from CLI flags).
type Config struct {
	Roots    string // comma-separated
	Min      string // e.g. "500MB"
	Type     string // all | dirs | files
	SkipForm bool
	Excludes []string
}

// App is the root model: form → scanning → tree.
type App struct {
	cfg           Config
	state         state
	values        *formValues
	form          *huh.Form
	scan          scanModel
	scanID        int
	tree          treeModel
	params        Params
	styles        styles
	width, height int
	err           error
}

// New builds the app. With SkipForm it validates cfg and starts scanning.
func New(cfg Config) (*App, error) {
	typ, err := tree.ParseType(cfg.Type)
	if err != nil {
		return nil, err
	}
	a := &App{cfg: cfg, values: &formValues{roots: cfg.Roots, min: cfg.Min, typ: typ}, styles: newStyles(true)}
	if cfg.SkipForm {
		p, err := a.values.params()
		if err != nil {
			return nil, err
		}
		a.startScan(p)
		return a, nil
	}
	a.showForm()
	return a, nil
}

func (a *App) Init() tea.Cmd {
	if a.state == stateScanning {
		return tea.Batch(tea.RequestBackgroundColor, a.scan.Init())
	}
	return tea.Batch(tea.RequestBackgroundColor, a.form.Init())
}

func (a *App) showForm() tea.Cmd {
	a.form = newForm(a.values)
	a.state = stateForm
	return a.form.Init()
}

func (a *App) startScan(p Params) tea.Cmd {
	a.params = p
	a.err = nil
	a.scanID++
	a.scan = newScanModel(a.scanID, p, a.cfg.Excludes, a.styles)
	a.state = stateScanning
	return a.scan.Init()
}

func (a *App) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		a.width, a.height = msg.Width, msg.Height
	case tea.BackgroundColorMsg:
		a.styles = newStyles(msg.IsDark())
		a.tree.styles = a.styles
		a.scan.styles = a.styles
	case tea.KeyPressMsg:
		if msg.String() == "ctrl+c" {
			if a.state == stateScanning {
				a.scan.cancel()
			}
			return a, tea.Quit
		}
	}

	switch a.state {
	case stateForm:
		m, cmd := a.form.Update(msg)
		a.form = m.(*huh.Form)
		switch a.form.State {
		case huh.StateCompleted:
			p, err := a.values.params()
			if err != nil {
				a.err = err
				return a, a.showForm()
			}
			return a, a.startScan(p)
		case huh.StateAborted:
			return a, tea.Quit
		}
		return a, cmd

	case stateScanning:
		switch msg := msg.(type) {
		case tea.KeyPressMsg:
			if msg.String() == "esc" {
				a.scan.cancel()
				return a, a.showForm()
			}
		case scanDoneMsg:
			if msg.id != a.scanID {
				return a, nil
			}
			if msg.err != nil {
				if !errors.Is(msg.err, context.Canceled) {
					a.err = msg.err
				}
				return a, a.showForm()
			}
			a.tree = newTreeModel(msg.res, a.params, a.styles, a.width, a.height)
			a.state = stateTree
			return a, nil
		}
		var cmd tea.Cmd
		a.scan, cmd = a.scan.Update(msg)
		return a, cmd

	case stateTree:
		switch msg.(type) {
		case rescanMsg:
			return a, a.startScan(a.params)
		case newSearchMsg:
			return a, a.showForm()
		}
		var cmd tea.Cmd
		a.tree, cmd = a.tree.Update(msg)
		return a, cmd
	}
	return a, nil
}

func (a *App) View() tea.View {
	var s string
	switch a.state {
	case stateForm:
		s = a.styles.title.Render("heft") + "  " + a.styles.subtle.Render("find big files and folders") + "\n\n"
		if a.err != nil {
			s += a.styles.err.Render(a.err.Error()) + "\n\n"
		}
		s += a.form.View()
	case stateScanning:
		s = a.scan.View(a.width)
	case stateTree:
		s = a.tree.View()
	}
	v := tea.NewView(s)
	v.AltScreen = true
	return v
}
```

```go
package main

import (
	"flag"
	"fmt"
	"os"
	"strings"

	tea "charm.land/bubbletea/v2"

	"github.com/david-potgieter/heft/internal/scan"
	"github.com/david-potgieter/heft/internal/ui"
)

var version = "dev" // set by goreleaser

func main() {
	minSize := flag.String("min", "100MB", "minimum size, e.g. 500MB, 1GiB")
	typ := flag.String("type", "all", "what to show: all, dirs, files")
	noForm := flag.Bool("no-form", false, "skip the form and scan immediately")
	showVersion := flag.Bool("version", false, "print version")
	flag.Usage = func() {
		fmt.Fprintf(os.Stderr, "Usage: heft [paths...] [flags]\n\nFind big files and folders.\n\n")
		flag.PrintDefaults()
	}

	// Allow flags before and after paths.
	var roots []string
	args := os.Args[1:]
	for {
		_ = flag.CommandLine.Parse(args)
		args = flag.Args()
		if len(args) == 0 {
			break
		}
		roots = append(roots, args[0])
		args = args[1:]
	}
	if *showVersion {
		fmt.Println("heft", version)
		return
	}
	if len(roots) == 0 {
		roots = []string{"~"}
	}

	home, _ := os.UserHomeDir()
	app, err := ui.New(ui.Config{
		Roots:    strings.Join(roots, ", "),
		Min:      *minSize,
		Type:     *typ,
		SkipForm: *noForm,
		Excludes: scan.DefaultExcludes(home),
	})
	if err != nil {
		fmt.Fprintln(os.Stderr, "heft:", err)
		os.Exit(2)
	}
	if _, err := tea.NewProgram(app).Run(); err != nil {
		fmt.Fprintln(os.Stderr, "heft:", err)
		os.Exit(1)
	}
}
```

````markdown
# heft

Find big files and folders on your Mac, in your terminal.

`heft` scans folders in parallel and shows everything over a size threshold as a navigable tree.
Built with [Bubble Tea](https://charm.land), [Bubbles](https://charm.land), [Lip Gloss](https://charm.land) and [Huh](https://charm.land).

## Install

```sh
go install github.com/david-potgieter/heft/cmd/heft@latest
```

## Usage

```sh
heft                                  # opens a form
heft ~/Work --min 500MB               # prefill the form
heft ~/Work ~/Downloads --min 1GB --type files --no-form
```

| flag | default | |
|---|---|---|
| `--min` | `100MB` | minimum size (`500MB`, `1GiB`, …) |
| `--type` | `all` | `all`, `dirs` or `files` |
| `--no-form` | `false` | scan immediately |

### Keys

| key | action |
|---|---|
| `↑`/`k` `↓`/`j` | move |
| `→`/`l`/`enter` | expand |
| `←`/`h` | collapse / go to parent |
| `g` / `G` | top / bottom |
| `s` | sort by size or name |
| `o` | reveal in Finder |
| `r` | rescan |
| `n` | new search |
| `?` | help |
| `q` | quit |

### Notes

- Sizes are actual disk usage. Hard links count once; symlinks aren't followed; other volumes are skipped.
- macOS protects some folders. Grant your terminal **Full Disk Access** (System Settings → Privacy & Security) for complete results.
````
